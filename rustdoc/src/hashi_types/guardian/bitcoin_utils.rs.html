101/// Validates that an unchecked address is valid for `network` and returns a checked address.
102fn validate_address_for_network(
103    address: &BitcoinAddress<NetworkUnchecked>,
104    network: Network,
105) -> super::GuardianResult<BitcoinAddress<NetworkChecked>> {
106    // Prefer the library's checked conversion to avoid accidentally assuming correctness.
107    address.clone().require_network(network).map_err(|_| {
108        InvalidInputs(format!(
109            "invalid address {:?} for network {}",
110            address, network
111        ))
112    })
113}

115/// Represents an input to be spent.
116///
117/// All inputs are expected to be P2TR (Pay-to-Taproot) since spending is done via taproot script path.
118impl InputUTXO {
119    /// Constructs a new `InputUTXO` and validates all invariants.
120    pub fn new(
121        outpoint: OutPoint,
122        amount: Amount,
123        address: BitcoinAddress<NetworkUnchecked>,
124        leaf_hash: TapLeafHash,
125        network: Network,
126    ) -> super::GuardianResult<Self> {
127        // TODO: Validate amount > 0.
128        let address = validate_address_for_network(&address, network)?;
129
130        if !address.script_pubkey().is_p2tr() {
131            return Err(InvalidInputs("input address is not p2tr".to_string()));
132        }
133
134        Ok(Self {
135            outpoint,
136            amount,
137            address,
138            leaf_hash,
139        })
140    }
141
142    pub fn from_wire(input: InputUTXOWire, network: Network) -> super::GuardianResult<Self> {
143        Self::new(
144            input.outpoint,
145            input.amount,
146            input.address,
147            input.leaf_hash,
148            network,
149        )
150    }
151
152    /// Returns a `TxIn` for this UTXO with placeholder witness data.
153    ///
154    /// The witness will be populated later after signing.
155    pub fn txin(&self) -> TxIn {
156        TxIn {
157            previous_output: self.outpoint,
158            // No script sig needed for taproot
159            script_sig: ScriptBuf::default(),
160            // Enables RBF, disables relative lock time, allows absolute lock time
161            sequence: Sequence::ENABLE_RBF_NO_LOCKTIME,
162            // Witness will be set later
163            witness: Witness::default(),
164        }
165    }
166
167    /// Returns the previous output as a `TxOut` (for sighash computation).
168    pub fn prevout(&self) -> TxOut {
169        TxOut {
170            value: self.amount,
171            script_pubkey: self.address.script_pubkey(),
172        }
173    }
174}

211/// Represents an output destination for a withdrawal.
212///
213/// Outputs can be **external** (to a user-provided address) or **internal** (change, derived inside enclave).
214impl OutputUTXO {
215    /// Constructs a new `OutputUTXO::External` variant.
216    pub fn new_external(
217        address: BitcoinAddress<NetworkUnchecked>,
218        amount: Amount,
219        network: Network,
220    ) -> super::GuardianResult<Self> {
221        Ok(OutputUTXO::External(ExternalOutputUTXO::new(
222            address, amount, network,
223        )?))
224    }
225
226    /// Constructs a new `OutputUTXO::Internal` variant.
227    pub fn new_internal(derivation_path: DerivationPath, amount: Amount) -> Self {
228        OutputUTXO::Internal(InternalOutputUTXO {
229            derivation_path,
230            amount,
231        })
232    }
233
234    pub fn from_wire(output: OutputUTXOWire, network: Network) -> super::GuardianResult<Self> {
235        Ok(match output {
236            OutputUTXOWire::External(external) => {
237                OutputUTXO::External(ExternalOutputUTXO::from_wire(external, network)?)
238            }
239            OutputUTXOWire::Internal(internal) => OutputUTXO::Internal(internal),
240        })
241    }
242
243    /// Returns the output amount in satoshis.
244    pub fn amount(&self) -> Amount {
245        match self {
246            OutputUTXO::External(ExternalOutputUTXO { amount, .. }) => *amount,
247            OutputUTXO::Internal(InternalOutputUTXO { amount, .. }) => *amount,
248        }
249    }
250
251    /// Constructs a `TxOut` for this output.
252    ///
253    /// `hashi_master_g` is the raw MPC verifying key (with y-parity preserved).
254    /// Internal outputs derive their child key from this raw G — using only
255    /// the x-only/even-y projection would silently produce a different child
256    /// key for half of all MPC vks, breaking the 2-of-2 leaf script.
257    pub fn to_txout(&self, enclave_pubkey: &BitcoinPubkey, hashi_master_g: &HashiMasterG) -> TxOut {
258        match self {
259            OutputUTXO::External(ExternalOutputUTXO { address, amount }) => TxOut {
260                value: *amount,
261                script_pubkey: address.script_pubkey(),
262            },
263            OutputUTXO::Internal(InternalOutputUTXO {
264                derivation_path,
265                amount,
266            }) => {
267                let scripts =
268                    compute_taproot_artifacts(enclave_pubkey, hashi_master_g, derivation_path);
269                TxOut {
270                    value: *amount,
271                    script_pubkey: scripts.0,
272                }
273            }
274        }
275    }
276}

278impl TxUTXOs {
279    /// Constructs a new `TxUTXOs` and validates all invariants.
280    pub fn new(inputs: Vec<InputUTXO>, outputs: Vec<OutputUTXO>) -> super::GuardianResult<Self> {
281        if inputs.is_empty() {
282            return Err(InvalidInputs("input utxos must not be empty".into()));
283        }
284        if outputs.is_empty() {
285            return Err(InvalidInputs("output utxos must not be empty".into()));
286        }
287
288        // Disallow duplicate inputs (same txid,vout), which would result in an invalid transaction.
289        let mut seen_inputs: HashSet<OutPoint> = HashSet::with_capacity(inputs.len());
290        for utxo in &inputs {
291            if !seen_inputs.insert(utxo.outpoint) {
292                return Err(InvalidInputs(format!(
293                    "duplicate input outpoint: {}",
294                    utxo.outpoint
295                )));
296            }
297        }
298
299        let tx_info = Self { inputs, outputs };
300
301        // Enforce the intended invariant: fees > 0.
302        tx_info.assert_positive_fees()?;
303
304        Ok(tx_info)
305    }
306
307    /// Returns a reference to the inputs.
308    pub fn get_inputs(&self) -> &[InputUTXO] {
309        &self.inputs
310    }
311
312    /// Returns a reference to the outputs.
313    pub fn get_outputs(&self) -> &[OutputUTXO] {
314        &self.outputs
315    }
316
317    /// Constructs all outputs (both external and internal).
318    ///
319    /// For `External` outputs, uses the user-provided address. For `Internal` outputs,
320    /// derives a taproot address using the enclave and hashi keys.
321    pub fn compute_all_outputs(
322        &self,
323        enclave_pubkey: &BitcoinPubkey,
324        hashi_master_g: &HashiMasterG,
325    ) -> Vec<TxOut> {
326        self.outputs
327            .iter()
328            .map(|utxo| utxo.to_txout(enclave_pubkey, hashi_master_g))
329            .collect()
330    }
331
332    pub fn external_outs(&self) -> Vec<&ExternalOutputUTXO> {
333        self.outputs
334            .iter()
335            .filter_map(|utxo| match utxo {
336                OutputUTXO::External(x) => Some(x),
337                OutputUTXO::Internal(_) => None,
338            })
339            .collect::<Vec<_>>()
340    }
341
342    pub fn external_out_amount(&self) -> Amount {
343        self.outputs
344            .iter()
345            .filter_map(|utxo| match utxo {
346                OutputUTXO::External(x) => Some(x.amount),
347                OutputUTXO::Internal(_) => None,
348            })
349            .sum()
350    }
351
352    /// BTC that leaves the pool when this txn broadcasts: `inputs - change`,
353    /// equivalent to `external_out_amount + miner_fee`. The amount that
354    /// consumes the rate-limiter (miner fee leaves the pool too; change
355    /// flows back).
356    pub fn gross_outflow_amount(&self) -> Amount {
357        let inputs: Amount = self.inputs.iter().map(|i| i.amount).sum();
358        let internal: Amount = self
359            .outputs
360            .iter()
361            .filter_map(|utxo| match utxo {
362                OutputUTXO::Internal(x) => Some(x.amount),
363                OutputUTXO::External(_) => None,
364            })
365            .sum();
366        inputs - internal
367    }
368
369    /// Constructs an unsigned Bitcoin transaction for these UTXOs.
370    fn unsigned_tx(
371        &self,
372        enclave_pubkey: &BitcoinPubkey,
373        hashi_master_g: &HashiMasterG,
374    ) -> Transaction {
375        let all_outputs = self.compute_all_outputs(enclave_pubkey, hashi_master_g);
376        construct_tx(
377            self.inputs.iter().map(|input| input.txin()).collect(),
378            all_outputs,
379        )
380    }
381
382    /// Constructs sighash messages for each input, ready for signing.
383    ///
384    /// Uses `taproot_script_spend_signature_hash` for script-path spending.
385    pub fn signing_messages_and_txid(
386        &self,
387        enclave_pubkey: &BitcoinPubkey,
388        hashi_master_g: &HashiMasterG,
389    ) -> (Vec<Message>, Txid) {
390        let tx = self.unsigned_tx(enclave_pubkey, hashi_master_g);
391        let prevouts: Vec<TxOut> = self.inputs.iter().map(|input| input.prevout()).collect();
392
393        let messages = self
394            .inputs
395            .iter()
396            .enumerate()
397            .map(|(index, input)| {
398                let mut sighasher = SighashCache::new(tx.clone());
399                let sighash = sighasher
400                    .taproot_script_spend_signature_hash(
401                        index,
402                        &Prevouts::All(&prevouts),
403                        input.leaf_hash,
404                        TapSighashType::Default,
405                    )
406                    .expect("sighash failed unexpectedly");
407                Message::from_digest(*sighash.as_byte_array())
408            })
409            .collect::<Vec<Message>>();
410        (messages, tx.compute_txid())
411    }
412
413    fn assert_positive_fees(&self) -> super::GuardianResult<()> {
414        let input_sum = self.inputs.iter().map(|utxo| utxo.amount).sum::<Amount>();
415        let output_sum = self.outputs.iter().map(|utxo| utxo.amount()).sum();
416        if input_sum <= output_sum {
417            return Err(InvalidInputs(format!(
418                "fees must be positive: input_sum={} output_sum={}",
419                input_sum, output_sum
420            )));
421        }
422        Ok(())
423    }
424}

430/// Signs messages using Schnorr signatures (suitable for taproot script-spend).
431///
432/// Each message is signed and wrapped in a `Signature` with `TapSighashType::Default`.
433pub fn sign_btc_tx(messages: &[Message], kp: &BitcoinKeypair) -> Vec<BitcoinSignature> {
434    messages
435        .iter()
436        // Not using aux randomness which only provides side-channel protection
437        .map(|m| BTC_LIB.sign_schnorr_no_aux_rand(m, kp))
438        .map(|s| Signature {
439            signature: s,
440            sighash_type: TapSighashType::Default,
441        })
442        .collect()
443}

445/// Constructs a Bitcoin transaction with the given inputs and outputs.
446///
447/// Uses BTC tx version 2 and disables lock time.
448pub fn construct_tx(inputs: Vec<TxIn>, outputs: Vec<TxOut>) -> Transaction {
449    Transaction {
450        // The latest BTC tx version
451        version: Version::TWO,
452        // Disable absolute lock time (i.e., can be mined immediately)
453        lock_time: LockTime::ZERO,
454        input: inputs,
455        output: outputs,
456    }

463/// Creates a taproot descriptor for the given enclave and hashi keys with a 2-of-2 multi_a script.
464/// Taproot addresses are constructed as follows:
465/// 1. Derive a child hashi pubkey from the derivation path
466/// 2. Create a 2-of-2 tapscript with the enclave key and derived hashi key
467/// 3. Place the tapscript as the sole leaf with a NUMS internal key
468///
469/// `hashi_master_g` is the raw MPC verifying key — the same `G` point that
470/// `fastcrypto_tbls::threshold_schnorr::key_derivation::derive_verifying_key`
471/// consumes inside MPC signing. Passing the x-only projection here would
472/// silently force an even-y parent and produce a derived child that doesn't
473/// match the MPC signature for ~half of all vks.
474pub fn compute_taproot_descriptor(
475    enclave_pubkey: &BitcoinPubkey,
476    hashi_master_g: &HashiMasterG,
477    hashi_derivation_path: &DerivationPath,
478) -> Tr<BitcoinPubkey> {
479    let derived_hashi_pubkey = derive_hashi_child_pubkey(hashi_master_g, hashi_derivation_path);
480
481    let internal = nums_internal_key();
482
483    // Taproot descriptor with one leaf: 2-of-2 checksigadd-style multisig
484    // Descriptor docs: https://github.com/bitcoin/bitcoin/blob/master/doc/descriptors.md
485    let desc_str = format!(
486        "tr({},multi_a(2,{},{}))",
487        internal, enclave_pubkey, derived_hashi_pubkey
488    );
489
490    match Descriptor::<BitcoinPubkey>::from_str(&desc_str).expect("valid descriptor") {
491        Descriptor::Tr(tr) => tr,
492        _ => panic!("unexpected descriptor"),
493    }
494}
495
496/// Derive the hashi child pubkey at `derivation_path` from the raw MPC
497/// verifying key `G` point. The result is the same x-only key that the MPC
498/// signing protocol produces signatures against, so the leaf script will
499/// agree with the on-chain signed witness.
500pub fn derive_hashi_child_pubkey(
501    hashi_master_g: &HashiMasterG,
502    hashi_derivation_path: &DerivationPath,
503) -> BitcoinPubkey {
504    let derived = derive_verifying_key(hashi_master_g, hashi_derivation_path);
505    BitcoinPubkey::from_slice(&derived.to_byte_array()).expect("derived schnorr key is x-only")
506}

514pub fn single_key_taproot_script_path_address(

522pub fn single_key_taproot_script_path_spend_artifacts(
523    pubkey: &BitcoinPubkey,
524) -> (ScriptBuf, ControlBlock, TapLeafHash) {
525    let (spend_info, tapscript) = single_key_taproot_script_path_spend_info(pubkey);
526    let leaf_version = LeafVersion::TapScript;
527    let control_block = spend_info
528        .control_block(&(tapscript.clone(), leaf_version))
529        .expect("control block exists for tapscript leaf");
530    let leaf_hash = TapLeafHash::from_script(&tapscript, leaf_version);
531    (tapscript, control_block, leaf_hash)
532}
533
534fn single_key_taproot_script_path_spend_info(
535    pubkey: &BitcoinPubkey,
536) -> (TaprootSpendInfo, ScriptBuf) {
537    let nums_internal_key = nums_internal_key();
538    let tapscript = script::Builder::new()
539        .push_x_only_key(pubkey)
540        .push_opcode(opcodes::all::OP_CHECKSIG)
541        .into_script();
542    let spend_info = TaprootBuilder::new()
543        .add_leaf(0, tapscript.clone())
544        .expect("valid single-leaf tapscript tree")
545        .finalize(&*BTC_LIB, nums_internal_key)
546        .expect("valid taproot spend info");
547    (spend_info, tapscript)
548}

550/// Computes both the address and leaf script for a given derivation path and network.
551fn compute_taproot_artifacts(
552    enclave_pubkey: &BitcoinPubkey,
553    hashi_master_g: &HashiMasterG,
554    hashi_derivation_path: &DerivationPath,
555) -> (ScriptBuf, TapLeafHash) {
556    let desc = compute_taproot_descriptor(enclave_pubkey, hashi_master_g, hashi_derivation_path);
557
558    let address_script = desc.script_pubkey();
559    let item = desc
560        .leaves()
561        .next()
562        .expect("tap tree should have at least one leaf");
563    let leaf_hash = item.compute_tap_leaf_hash();
564
565    (address_script, leaf_hash)
566}

568/// Deposit address for `tr(NUMS, multi_a(2, enclave, derived_hashi))`.
569pub fn two_of_two_taproot_script_path_address(
570    enclave_pubkey: &BitcoinPubkey,
571    hashi_master_g: &HashiMasterG,
572    hashi_derivation_path: &DerivationPath,
573    network: Network,
574) -> BitcoinAddress {
575    compute_taproot_descriptor(enclave_pubkey, hashi_master_g, hashi_derivation_path)
576        .address(network)
577}
578
579/// 2-of-2 sibling of [`single_key_taproot_script_path_spend_artifacts`].
580pub fn two_of_two_taproot_script_path_spend_artifacts(
581    enclave_pubkey: &BitcoinPubkey,
582    hashi_master_g: &HashiMasterG,
583    hashi_derivation_path: &DerivationPath,
584) -> (ScriptBuf, ControlBlock, TapLeafHash) {
585    let desc = compute_taproot_descriptor(enclave_pubkey, hashi_master_g, hashi_derivation_path);
586
587    let tap_tree = desc.tap_tree().expect("descriptor should have a tap tree");
588    let leaf = tap_tree
589        .leaves()
590        .next()
591        .expect("tap tree should have at least one leaf");
592    let tap_script = leaf.compute_script();
593
594    let control_block = desc
595        .spend_info()
596        .leaves()
597        .next()
598        .expect("spend info should have at least one leaf")
599        .into_control_block();
600
601    let leaf_hash = TapLeafHash::from_script(&tap_script, LeafVersion::TapScript);
602
603    (tap_script, control_block, leaf_hash)
604}

699    fn gen_keypair_and_address(
700        bytes: Option<[u8; 32]>,
701        network: Network,
702    ) -> (BitcoinKeypair, BitcoinAddress) {
703        let mut rng = rand::thread_rng();
704        let bytes = bytes.unwrap_or({
705            let mut bytes = [0u8; 32];
706            rand::Rng::fill(&mut rng, &mut bytes);
707            bytes
708        });
709        let keypair = create_btc_keypair(&bytes);
710        let (internal_key, _) = UntweakedPublicKey::from_keypair(&keypair);
711        let address = BitcoinAddress::p2tr(&BTC_LIB, internal_key, None, network);
712        (keypair, address)
713    }

715    fn construct_witness(
716        hashi_signature: &Signature,
717        enclave_signature: &Signature,
718        script: &ScriptBuf,
719        control_block: &ControlBlock,
720    ) -> Witness {
721        // Witness stack order: [sig_for_pk2, sig_for_pk1, script, control_block]
722        // Since our script is <pk1> OP_CHECKSIG <pk2> OP_CHECKSIGADD ...
723        // And stack is LIFO, we need: [hashi_sig, enclave_sig, script, control]
724        let hashi_sig_vec = hashi_signature.to_vec();
725        let enclave_sig_vec = enclave_signature.to_vec();
726        let control_block_vec = control_block.serialize();
727        let witness_elements: Vec<Vec<u8>> = vec![
728            hashi_sig_vec,     // sig for pk2 (hashi)
729            enclave_sig_vec,   // sig for pk1 (enclave)
730            script.to_bytes(), // script
731            control_block_vec, // control block
732        ];
733        Witness::from_slice(&witness_elements)
734    }

736    /// Convert a bitcoin-lib x-only key into the even-y `G` point. The
737    /// bitcoin-lib `Keypair` always signs against the even-y projection of
738    /// its master pubkey, so the descriptor derivation must use the same
739    /// parent to agree.
740    fn hashi_master_g_from_xonly(pubkey: &BitcoinPubkey) -> HashiMasterG {
741        HashiMasterG::with_even_y_from_x_be_bytes(&pubkey.serialize()).expect("valid x coordinate")
742    }

822
823        let enclave_pk = enclave_keypair.x_only_public_key().0;
824        let hashi_pk = hashi_keypair.x_only_public_key().0;
825
826        let (address, control_block, tap_script) =
827            create_taproot_artifacts_for_test(&enclave_pk, &hashi_pk, &[0u8; 32], Regtest);
828        println!("\n=== 2-of-2 Multisig Address ===");
829        println!("Address: {}", address);
830        println!("Enclave pubkey: {}", enclave_pk);
831        println!("Hashi pubkey: {}", hashi_pk);
832
833        // A) User picks destination address.
834        const DEST_SK: [u8; 32] = [3u8; 32];
835        let (_, dest_address) = gen_keypair_and_address(Some(DEST_SK), Regtest);
836
837        // B) Hashi selects a UTXO
838        // NOTE: Paste a real regtest UTXO to obtain a broadcastable tx.
839        let out_point = OutPoint {
840            txid: "f62f8d94074084555bd28187a4c79648c72571e53b5e2ba823bdf92b2cc1f88c"
841                .parse()
842                .unwrap(),
843            vout: 1,
844        };
845        let hashi_master_g = hashi_master_g_from_xonly(&hashi_pk);
846        let (_, leaf_hash) = compute_taproot_artifacts(&enclave_pk, &hashi_master_g, &[0u8; 32]);
847
848        let input_amount = Amount::from_sat(100000000); // 1.0 BTC
849        let input_utxo = InputUTXO::new(
850            out_point,
851            input_amount,
852            address.as_unchecked().clone(),
853            leaf_hash,
854            Regtest,
855        )
856        .unwrap();
857
858        // C) Enclave signs the transaction.
859        let tx_info = TxUTXOs::new(
860            vec![input_utxo.clone()],
861            vec![
862                OutputUTXO::External(ExternalOutputUTXO {
863                    address: dest_address,
864                    amount: Amount::from_sat(100), // 100 sats is sent
865                }),
866                OutputUTXO::Internal(InternalOutputUTXO {
867                    derivation_path: [0; 32],
868                    amount: input_amount - Amount::from_sat(1000),
869                }),
870            ],
871        )
872        .unwrap();
873
874        let (messages, _) = tx_info.signing_messages_and_txid(&enclave_pk, &hashi_master_g);
875        let enclave_signatures = sign_btc_tx(&messages, &enclave_keypair);
876
877        // D) Hashi signs the transaction.
878        let hashi_signatures = sign_btc_tx(&messages, &hashi_keypair);
879
880        // E) Relayer combines the signatures and finalizes the transaction.
881        // Note: If there are multiple inputs, we need to construct the witness for each input.
882        assert_eq!(enclave_signatures.len(), 1);
883        assert_eq!(hashi_signatures.len(), 1);
884        let witness = construct_witness(
885            &hashi_signatures[0],
886            &enclave_signatures[0],
887            &tap_script,
888            &control_block,
889        );
890
891        let mut input_txin = input_utxo.txin();
892        input_txin.witness = witness;
893
894        let all_outputs = tx_info.compute_all_outputs(&enclave_pk, &hashi_master_g);
895        let signed_tx = construct_tx(vec![input_txin], all_outputs);
896        println!("Signed TX: {:#?}", signed_tx);
897        println!("TXID: {}", signed_tx.compute_txid());
898        println!(
899            "Transaction hex: {}",
900            consensus::encode::serialize_hex(&signed_tx)
901        );
902    }

904    // Regression: an odd-y MPC master key used to silently disagree between
905    // the 2-of-2 leaf script (which reconstructed an even-y parent from the
906    // x-only bytes) and the MPC signature (which signs against
907    // `derive_verifying_key(raw_g, path)`), giving "Invalid Schnorr signature"
908    // at Bitcoin. This builds the real leaf for an odd-y master and asserts it
909    // embeds the MPC-signed child key, not the even-y-forced one.
910    #[test]
911    fn two_of_two_leaf_embeds_mpc_signed_child_for_odd_y_master() {
912        use fastcrypto::groups::GroupElement;
913        use fastcrypto_tbls::threshold_schnorr::S;
914
915        // Find a master G with ODD y so we exercise the formerly-broken branch.
916        let raw_g = loop {
917            let mut bytes = [0u8; 32];
918            rand::Rng::fill(&mut rand::thread_rng(), &mut bytes);
919            let g = HashiMasterG::generator() * S::from_bytes_mod_order(&bytes);
920            if !g.has_even_y().unwrap() {
921                break g;
922            }
923        };
924
925        let enclave_pubkey = create_btc_keypair(&[7u8; 32]).x_only_public_key().0;
926        let path = [42u8; 32];
927
928        // The child key the MPC actually produces signatures against.
929        let mpc_child = derive_verifying_key(&raw_g, &path).to_byte_array();
930        // The child the old even-y-forcing code would have embedded instead.
931        let raw_g_xonly = SchnorrPublicKey::try_from(&raw_g)
932            .expect("valid schnorr key")
933            .to_byte_array();
934        let forced_even =
935            HashiMasterG::with_even_y_from_x_be_bytes(&raw_g_xonly).expect("valid x coordinate");
936        let buggy_child = derive_verifying_key(&forced_even, &path).to_byte_array();
937        assert_ne!(
938            buggy_child, mpc_child,
939            "odd-y parent must change the child, else this test can't catch the regression"
940        );
941
942        // The production 2-of-2 leaf must embed the MPC-signed child.
943        let (leaf_script, _, _) =
944            two_of_two_taproot_script_path_spend_artifacts(&enclave_pubkey, &raw_g, &path);
945        let script = leaf_script.as_bytes();
946        assert!(
947            script.windows(32).any(|w| w == mpc_child.as_slice()),
948            "2-of-2 leaf must embed the MPC-signed child key"
949        );
950        assert!(
951            !script.windows(32).any(|w| w == buggy_child.as_slice()),
952            "2-of-2 leaf must not embed the even-y-forced child key"
953        );
954    }
