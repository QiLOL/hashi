20/// The withdraw-mode state to install, built from `WithdrawModeConfig` (incl. the
21/// computed `state_hash` and the constructed rate limiter).
22pub(crate) struct WithdrawInstall {
23    state_hash: [u8; 32],
24    network: bitcoin::Network,
25    secret_sharing_instance: SecretSharingInstance,
26    hashi_btc_master_pubkey: HashiMasterG,
27    committee: HashiCommittee,
28    rate_limiter: RateLimiter,
29}
30
31impl WithdrawInstall {
32    /// Decompose a `WithdrawModeConfig` into the install bundle, constructing the
33    /// rate limiter (the only fallible step). Shared by `operator_init` and tests.
34    pub(crate) fn from_config(config: WithdrawModeConfig) -> GuardianResult<Self> {
35        let (withdraw_state, secret_sharing_instance, network) = config.into_parts();
36        let state_hash = withdraw_state.digest();
37        let (committee, limiter_config, limiter_state, hashi_btc_master_pubkey) =
38            withdraw_state.into_parts();
39        let rate_limiter = RateLimiter::new(limiter_config, limiter_state)?;
40        Ok(Self {
41            state_hash,
42            network,
43            secret_sharing_instance,
44            hashi_btc_master_pubkey,
45            committee,
46            rate_limiter,
47        })
48    }
49
50    /// Install the bundle onto a fresh enclave. Infallible by design (see the
51    /// `operator_init` invariant): every set runs once on a fresh enclave.
52    pub(crate) fn install_into(self, enclave: &Enclave) {
53        info!("Setting bitcoin network to {:?}.", self.network);
54        enclave
55            .config
56            .set_bitcoin_network(self.network)
57            .expect("Unable to set network");
58
59        info!(
60            "Storing secret-sharing instance: n={}, t={}, {} commitments.",
61            self.secret_sharing_instance.num_shares(),
62            self.secret_sharing_instance.threshold(),
63            self.secret_sharing_instance.commitments().len()
64        );
65        enclave
66            .set_secret_sharing_instance(self.secret_sharing_instance)
67            .expect("Unable to set secret-sharing instance");
68
69        info!("Setting state hash.");
70        enclave
71            .set_state_hash(self.state_hash)
72            .expect("Unable to set state hash");
73
74        info!("Setting hashi BTC master pubkey.");
75        enclave
76            .config
77            .set_hashi_btc_pk(self.hashi_btc_master_pubkey)
78            .expect("Unable to set hashi BTC master pubkey");
79
80        info!("Installing committee and rate limiter.");
81        enclave
82            .state
83            .init(self.committee, self.rate_limiter)
84            .expect("Unable to init enclave state");
85    }
86}

88/// Receives S3 API keys and — for a withdraw-mode enclave — the `WithdrawModeConfig`
89/// (committee, limiter, BTC master pubkey, secret-sharing instance, network);
90/// installs them and fixes the `state_hash`. A ceremony enclave carries only S3.
91///
92/// Invariant: operator_init never returns an `Err` from a partially-initialized
93/// enclave. Every fallible step (request validation, S3 connectivity, rate-limiter
94/// construction) runs before any state is mutated, so an early `Err` leaves the
95/// enclave untouched and retryable. The mutation then happens entirely in
96/// `commit_operator_init`, which returns `()` — it cannot report an error, so a
97/// half-mutated enclave is never observed via an `Err`.
98pub async fn operator_init(
99    enclave: Arc<Enclave>,
100    request: OperatorInitRequest,
101) -> GuardianResult<()> {
102    info!("/operator_init - Received request.");
103
104    // Serialize so concurrent callers can't race the check-then-commit below.
105    let _guard = enclave.control_lock.lock().await;
106
107    if enclave.is_operator_init_complete() {
108        return Err(InvalidInputs("operator_init already complete".into()));
109    }
110
111    // ---- Validate & build: Nothing in this phase mutates enclave state, so any
112    // error here leaves the enclave untouched. ----
113
114    // A withdraw-mode enclave must carry the config and a ceremony enclave must not.
115    request.validate(enclave.mode())?;
116
117    let (s3_config, state) = request.into_parts();
118    let logger = S3Logger::new_checked(&s3_config).await?;
119    info!("S3 connectivity check complete.");
120
121    // Build the withdraw-mode install bundle (incl. the rate limiter, the last
122    // fallible step) up front; `None` for a ceremony enclave.
123    let withdraw = match state {
124        Some(config) => Some(WithdrawInstall::from_config(config)?),
125        None => None,
126    };
127
128    // ---- All-or-nothing Commit: Nothing in this phase errors out. ----
129    commit_operator_init(&enclave, logger, withdraw).await;
130    Ok(())
131}

133/// Install the validated config on the enclave and write the operator_init logs.
134/// Infallible by design (returns `()`, see the `operator_init` invariant): every
135/// `set` here runs on a fresh enclave under the control lock, and the I/O steps
136/// (attestation, S3 logging) panic on failure rather than return.
137async fn commit_operator_init(
138    enclave: &Enclave,
139    logger: S3Logger,
140    withdraw: Option<WithdrawInstall>,
141) {
142    info!("Storing S3 configuration.");
143    enclave
144        .config
145        .set_s3_logger(logger)
146        .expect("Unable to set logger");
147
148    // Withdraw-mode state (committee, limiter, BTC master pubkey, instance, network,
149    // state_hash); a ceremony enclave installs none of it.
150    if let Some(install) = withdraw {
151        install.install_into(enclave);
152    }
153
154    // Log to S3!
155    // 1) Attestation and pub key help authenticate all subsequent enclave-signed messages.
156    let signing_pk = enclave.signing_pubkey();
157    enclave
158        .log_init(OIAttestationUnsigned {
159            attestation: get_attestation(&signing_pk).expect("Unable to get attestation"),
160            signing_public_key: signing_pk,
161        })
162        .await
163        .expect("Unable to log OperatorInitAttestationUnsigned");
164
165    // 2) Share commitments help KPs confirm that the right private key will be constructed.
166    enclave
167        .log_init(OIGuardianInfo(enclave.info()))
168        .await
169        .expect("Unable to log GuardianInfo");
170
171    enclave
172        .scratchpad
173        .operator_init_logging_complete
174        .set(())
175        .expect("operator_init_logging_complete should only be set once");
176
177    info!("Operator initialization complete.");
178}

180/// Receives one KP's encrypted share. Decrypts it under the enclave's state_hash
181/// (set at operator_init) as AAD — so only shares from KPs that agreed on the
182/// operator-supplied state decrypt — verifies it against the commitments, and
183/// reconstructs the BTC key once threshold shares arrive.
184pub async fn provisioner_init(
185    enclave: Arc<Enclave>,
186    request: ProvisionerInitRequest,
187) -> GuardianResult<()> {
188    info!("/provisioner_init - Received request.");
189
190    // Ensure only one provisioner_init request runs at a time to keep things simple.
191    // We reuse the decrypted_shares mutex lock for this purpose.
192    let mut received_shares = enclave.decrypted_shares().lock().await;
193
194    // Validation
195    if !enclave.is_operator_init_complete() {
196        return Err(InvalidInputs("Do operator init first".into()));
197    }
198    if enclave.is_provisioner_init_complete() {
199        return Err(InvalidInputs("Provisioner init already complete".into()));
200    }
201    info!("Enclave state validated.");
202
203    let sk = enclave.encryption_secret_key();
204    let share_id = request.encrypted_share().id;
205    // Always set here: provisioner_init is withdraw-mode only, and the
206    // operator_init check above guarantees a withdraw-mode enclave installed it.
207    let state_hash = *enclave
208        .state_hash()
209        .expect("withdraw-mode operator_init installs the state_hash");
210    info!("Share ID: {:?}.", share_id);
211
212    // 1) Decrypt the share (AAD = enclave state_hash). A share only decrypts if
213    //    the KP bound the same state the operator configured.
214    info!("Decrypting share.");
215    let share = decrypt_share(request.encrypted_share(), sk, Some(&state_hash))?;
216    info!("Share decrypted.");
217
218    // 2) Verify the share against the commitment
219    info!("Verifying share against commitment.");
220    let instance = enclave
221        .secret_sharing_instance()
222        .expect("secret-sharing instance should be set after operator_init");
223    instance.commitments().verify_share(&share)?;
224    info!("Share verified.");
225
226    // MILESTONE: a share that decrypts under the enclave state_hash and matches a
227    // commitment is a legitimate submission from a KP that agreed on the state.
228
229    // 3) Persist share
230    info!("Persisting share.");
231    let share_id = share.id;
232    // Check for duplicate share ID (linear search is fine for small share count)
233    if received_shares.iter().any(|s| s.id == share_id) {
234        return Err(InvalidInputs("Duplicate share ID".into()));
235    }
236    received_shares.push(share);
237    let current_share_count = received_shares.len();
238    let threshold = instance.threshold();
239    info!("Total shares received: {current_share_count}/{threshold}.");
240
241    // Note: This S3 log does not serve any security purpose.
242    enclave
243        .log_init(PISuccess {
244            share_id,
245            state_hash,
246        })
247        .await
248        .expect("Unable to log ProvisionerInitSuccess");
249
250    // 4) If we have enough shares, reconstruct the BTC key & finish initialization.
251    if current_share_count >= threshold {
252        let shares_vec: Vec<Share> = received_shares.iter().cloned().collect();
253        finalize_init(&shares_vec, threshold, &enclave).await;
254        // Log to S3 indicating that withdrawals can be expected henceforth
255        enclave
256            .log_init(PIEnclaveFullyInitialized)
257            .await
258            .expect("Unable to log EnclaveFullyInitialized");
259
260        // Clear shares as we are done using them
261        received_shares.clear();
262        enclave
263            .scratchpad
264            .provisioner_init_logging_complete
265            .set(())
266            .expect("provisioner_init_logging_complete should only be set once");
267    }
268
269    Ok(())
270}

272/// Reconstruct the BTC key from the threshold shares and install it. The rest of
273/// the enclave state was set at operator_init.
274/// Panics upon an error as the enclaves state is irrecoverable at this point.
275async fn finalize_init(shares: &[Share], threshold: usize, enclave: &Arc<Enclave>) {
276    info!("Threshold reached, combining shares.");
277    let enclave_k256_sk = combine_shares(shares, threshold).expect("Unable to combine shares");
278    let enclave_btc_keypair = k256_sk_to_btc_keypair(&enclave_k256_sk);
279
280    info!("Setting enclave keypair.");
281    enclave
282        .config
283        .set_btc_keypair(enclave_btc_keypair)
284        .expect("Unable to set enclave keypair");
285
286    info!("Enclave initialization complete.");
287}

312    /// Run commit_operator_init on a fresh enclave for the given mode (withdraw =>
313    /// carries the WithdrawModeConfig install bundle; ceremony => none).
314    async fn commit_for_mode(mode: EnclaveMode) -> Arc<Enclave> {
315        let enclave = Arc::new(Enclave::new(
316            GuardianSignKeyPair::new(rand::thread_rng()),
317            GuardianEncKeyPair::random(&mut rand::thread_rng()),
318            mode,
319        ));
320
321        let withdraw = match mode {
322            EnclaveMode::Withdraw => Some(
323                WithdrawInstall::from_config(WithdrawModeConfig::mock_for_testing(None)).unwrap(),
324            ),
325            EnclaveMode::Ceremony => None,
326        };
327
328        commit_operator_init(&enclave, crate::test_utils::mock_logger(), withdraw).await;
329        enclave
330    }

344    #[tokio::test]
345    async fn test_provisioner_init() {
346        let (shares, enclave) = setup_test_shares_and_enclave().await;
347        let state_hash = *enclave.state_hash().unwrap();
348
349        // Simulate KPs submitting share-only requests, each bound to the
350        // enclave's state_hash.
351        for (i, share) in shares.iter().enumerate().take(TEST_N) {
352            let request = ProvisionerInitRequest::build_from_share(
353                share,
354                enclave.encryption_public_key(),
355                state_hash,
356                &mut rand::thread_rng(),
357            );
358
359            let result = provisioner_init(enclave.clone(), request).await;
360
361            if i == TEST_T - 1 {
362                // At exactly threshold (first time), the BTC key is reconstructed.
363                assert!(
364                    result.is_ok(),
365                    "Should succeed at threshold (iteration {i})"
366                );
367                assert!(
368                    enclave.config.is_enclave_btc_keypair_set(),
369                    "Bitcoin key should be set after threshold"
370                );
371                assert!(
372                    enclave.is_fully_initialized(),
373                    "fully initialized at threshold"
374                );
375            } else if i >= TEST_T {
376                // After threshold, subsequent calls fail (already complete).
377                assert!(result.is_err(), "Should fail at iteration {i}: {result:?}");
378                assert!(
379                    enclave.config.is_enclave_btc_keypair_set(),
380                    "Bitcoin key should still be set"
381                );
382            } else {
383                // Before threshold, the BTC key is not yet reconstructed.
384                assert!(result.is_ok(), "Init should succeed before threshold");
385                assert!(
386                    !enclave.config.is_enclave_btc_keypair_set(),
387                    "Bitcoin key should not be set before threshold"
388                );
389            }
390        }
391    }
